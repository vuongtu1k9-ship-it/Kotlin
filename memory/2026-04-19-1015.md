private fun DrawScope.drawPieces(board: Board, selectedPiece: Piece?) {
    val cellSize = size.width / 9f
    val pieceRadius = cellSize / 2.5f

    board.getPieces().forEach { piece ->
        val centerX = piece.x * cellSize + cellSize / 2
        val centerY = piece.y * cellSize + cellSize / 2

        // Vẽ nền quân cờ (vàng cho Đỏ, xanh ngọc cho Đen)
        drawCircle(
            color = if (piece.color == PieceColor.RED) Color(0xFFFFD700) else Color(0xFF20B2AA), // Vàng / Xanh ngọc
            radius = pieceRadius,
            center = Offset(centerX, centerY)
        )

        // Vẽ viền quân cờ
        drawCircle(
            color = Color.Black,
            radius = pieceRadius,
            center = Offset(centerX, centerY),
            style = androidx.compose.ui.graphics.drawscope.Stroke(width = 2f)
        )

        // Đánh dấu quân cờ được chọn
        if (selectedPiece == piece) {
            drawCircle(
                color = Color.Yellow.copy(alpha = 0.5f),
                radius = pieceRadius * 1.2f,
                center = Offset(centerX, centerY)
            )
        }

        // Vẽ tên quân cờ (chữ Hán)
        val pieceText = when (piece.type) {
            PieceType.GENERAL -> if (piece.color == PieceColor.RED) "帥" else "將"  // Tướng
            PieceType.ADVISOR -> "士"   // Sĩ
            PieceType.ELEPHANT -> "象" // Tượng
            PieceType.CHARIOT -> "車"  // Xe
            PieceType.CANNON -> "炮"   // Pháo
            PieceType.HORSE -> "馬"    // Mã
            PieceType.SOLDIER -> if (piece.color == PieceColor.RED) "兵" else "卒"  // Tốt
        }

        // Màu chữ: Đỏ cho quân Đỏ, Đen cho quân Đen
        drawContext.canvas.nativeCanvas.drawText(
            pieceText,
            centerX,
            centerY,
            android.graphics.Paint().apply {
                textSize = pieceRadius * 1.2f
                color = if (piece.color == PieceColor.RED) android.graphics.Color.RED else android.graphics.Color.BLACK
                textAlign = android.graphics.Paint.Align.CENTER
                isFakeBoldText = true
                // Thêm font chữ Hán (nếu có)
                try {
                    typeface = android.graphics.Typeface.createFromAsset(
                        context.assets,
                        "font/NotoSansCJK-Regular.ttf"
                    )
                } catch (e: Exception) {
                    // Sử dụng font mặc định nếu không tìm thấy
                }
            }
        )
    }
}
